const progress = () => page.evaluate(() => {
    const game = globalThis.__survivalGame;
    return { day: game.day, supplies: game.supplies, phase: game.phase,
        current: game.currentJournal, entries: game.journalEntries, milestones: game.milestones };
});
const visibleText = () => page.evaluate(() => {
    const root = cc.director.getScene().getChildByName('Canvas').getChildByName('GameRoot');
    return Object.fromEntries(root.getComponentsInChildren('UIValue')
        .filter(binding => binding.node.activeInHierarchy)
        .map(binding => [binding.key, binding.node.getComponent(cc.Label)?.string ?? '']));
});
const startFormal = async (button) => {
    await click(button);
    assert((await state()).overlay === 'entry', 'Departure tier chooser did not open');
    await click('entry-tier-emergency');
    await click('entry-confirm');
    assert((await state()).phase === 'map', 'Departure did not enter map');
};
const fatalBattle = async () => {
    await page.evaluate(() => {
        const game = globalThis.__survivalGame;
        Object.assign(game, { fixedDone: true, route: 'A', pathIndex: 0 });
        const result = game.enterNode();
        if (!result.ok) throw new Error(result.message);
        Object.assign(game, { armor: 0, critical: true });
        Object.assign(game.battle, { block: 0, intent: 'attack' });
        game.revision++;
    });
    await page.waitForFunction(() => globalThis.__emberUI.snapshot().hits.some(hit => hit.id === 'end-turn' && hit.enabled));
    await click('end-turn');
    assert((await state()).phase === 'failure', 'Fatal enemy turn did not show failure');
};
const finishPreparedMap = async (supplies, armor = 10) => {
    await page.evaluate(({ supplies, armor }) => {
        const game = globalThis.__survivalGame;
        Object.assign(game, { fixedDone: true, route: 'A', pathIndex: game.scenario.routes.A.length,
            visited: ['fixed', ...game.scenario.routes.A], supplies, armor, returnElapsed: 0 });
        game.revision++;
    }, { supplies, armor });
    await page.waitForFunction(() => ['journal', 'shelter'].includes(globalThis.__emberUI.snapshot().phase));
    await idle();
};
async function firstClear(attempt, save, bank, armor = 10, ruleset = '') {
    await page.goto(`http://127.0.0.1:7457/?seed=17&save=${save ? 'on' : 'off'}&qa=core-loop-${attempt}&ruleset=${ruleset}`);
    await idle();
    await startFormal('journal-start');
    for (let index = 1; index < attempt; index++) {
        await fatalBattle();
        let result = await progress();
        assert(result.day === index + 1 && result.entries.length === 1 && result.current === null,
            'Ordinary failure changed journal progress or did not advance exactly one day');
        assert((await state()).supplies === 10 && (await state()).armor === 10, 'Failure recovery changed');
        await click('journal-start');
        assert((await state()).phase === 'shelter', 'Retry did not return to shelter');
        await startFormal('shelter-done');
    }
    await finishPreparedMap(bank, armor);
    const result = await progress(), labels = await visibleText();
    assert(result.current?.id === 'street-cleared' && result.current.day === attempt + 1, 'First clear has wrong journal date');
    assert(labels['journal/text1'] === `第 ${attempt + 1} 天 · ${result.current.title}`, 'Native journal title did not bind actual date');
    assert(labels['journal/text2'] === result.current.body, 'Native journal body differs from configured story');
    assert(result.supplies === bank, 'Returning charged a daily supply fee');
    await screenshot(`core-loop-day-${attempt + 1}`);
    return result.current;
}

const mapViews = async () => page.evaluate(() => {
    const root = cc.director.getScene().getChildByName('Canvas').getChildByName('GameRoot');
    return root.getComponentsInChildren('RouteMap').filter(map => map.node.activeInHierarchy).map(map => ({
        ...map.inspect(),
        labels: map.content.getComponentsInChildren(cc.Label).filter(label => label.node.activeInHierarchy).map(label => label.string),
        nativePoints: map.content.getComponentsInChildren('RoutePoint').filter(view => view.node.activeInHierarchy).map(view => ({
            id: view.data.id,
            enabled: view.action.getComponent(cc.Button).interactable,
            titleActive: view.title?.node.activeInHierarchy ?? false,
            statusActive: view.status?.node.activeInHierarchy ?? false,
        })),
    }));
});
const checkWindow = async (layer, readonly = false) => {
    const maps = await mapViews();
    assert(maps.length > 0, 'No native map is visible');
    for (const map of maps) {
        assert(map.currentLayer === layer, 'Wrong current layer: ' + JSON.stringify(map));
        assert(map.points.every(point => point.layer === layer || point.layer === layer + 1), 'A hidden layer leaked');
        const ids = new Set(map.points.map(point => point.id));
        assert(map.edges.every(edge => ids.has(edge.from) && ids.has(edge.to)), 'An edge reveals a hidden layer');
        assert(map.nativePoints.length === map.points.length, 'Old native points remain visible');
        assert(map.nativePoints.every(point => !point.titleActive && !point.statusActive), 'A point label is visible');
        assert(map.labels.length === 0, 'Map nodes contain visible text: ' + map.labels.join(','));
        if (readonly) assert(map.nativePoints.every(point => !point.enabled), 'Read-only map can move the player');
    }
    return maps[0];
};
const chooseOpening = async (side) => {
    const before = await state();
    const choices = before.hits.filter(hit => hit.id === 'enter-fixed' && hit.enabled).sort((a, b) => a.x - b.x);
    assert(choices.length === 2, 'There must be two clickable opening icons');
    const hit = choices[side], p = await point(hit.x + hit.width / 2, hit.y + hit.height / 2);
    await page.mouse.move(p.x, p.y);
    await page.waitForTimeout(120);
    assert((await mapViews())[0].labels.length === 0, 'Hover showed a node label');
    await page.mouse.click(p.x, p.y);
    await idle();
    trace.push({ action: side ? 'right-opening-icon' : 'left-opening-icon' });
    assert((await state()).phase === 'rewards', 'Opening icon did not enter the supply encounter');
    return page.evaluate(() => globalThis.__survivalGame.exportState());
};

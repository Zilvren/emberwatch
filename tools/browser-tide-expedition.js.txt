async function battleToMap() {
    for (let step=0;step<110;step++) {
        const s=await state();
        if (s.phase!=='battle') { assert(s.phase==='map','Combat ended in '+s.phase); return; }
        if ((await discovery()).length) {
            const mid=await checkpoint();
            assert(mid.battle.coreExposed,'Unexpected second opening offer.');
            await page.reload();await idle();
            assert(JSON.stringify(await checkpoint())===JSON.stringify(mid),'Mid-turn discovery did not resume exactly.');
            await click('reward-select-0');await screenshot('08b-exposed-core');await click('discover-confirm');
            assert((await state()).stamina===mid.battle.stamina,'Taking a mid-turn discovery refilled stamina.');
            continue;
        }
        const move=await page.evaluate(()=>{
            const g=globalThis.__survivalGame,b=g.battle;
            const cards=b.hand.filter(c=>!g.cardBlocked(c));
            const def=c=>g.content.cards[c.kind];
            const damage=cards.filter(c=>def(c).effect.type==='damage').sort((a,b)=>g.attackDamage(b.kind)-g.attackDamage(a.kind));
            let pick=damage.find(c=>g.attackDamage(c.kind)>=b.hp) || cards.find(c=>def(c).cost===0&&
                (def(c).effect.type==='damage'||(def(c).effect.type==='block'&&b.block<g.incomingDamage)));
            if (!pick && b.block<g.incomingDamage && g.health<7) pick=cards.find(c=>def(c).effect.type==='block');
            pick=pick||damage[0]||cards.find(c=>def(c).effect.type==='draw'||def(c).effect.type==='equip'||
                (def(c).effect.type==='block'&&b.block<g.incomingDamage));
            return pick ? {id:pick.id,attack:def(pick).effect.type==='damage'} : null;
        });
        if (!move) await click('end-turn');
        else {await click('card-'+move.id);if(move.attack)await click('enemy-target');}
    }
    throw Error('Combat exceeded the action guard.');
}
